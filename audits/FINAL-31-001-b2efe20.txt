Trước mỗi model call, tool call, tác động và chuyển trạng thái của worker, adapter phải kiểm tra fence/revision, binding tin cậy mới, thu hồi/hết hạn, nguồn, quyền, phê duyệt, quota/chi phí, trạng thái và kill switch. Lượt đã đặt trước tính cả khi lỗi hoặc không có phản hồi, không hoàn lại hoặc đặt lại khi retry. Kiểm tra trùng mà không gọi không tiêu đơn vị. Nếu tool vừa tính hạn mức vừa tạo tác động, adapter phải giữ cả điều kiện đặt trước và điều kiện chống lặp/fence tại ranh giới gọi, mỗi đặt trước chỉ dùng một lần. Ledger phải bền vững và kiểm tra được khi có nhiều tiến trình.

Phát hiện lease hết hạn cả khi worker im lặng. Khi BLOCKED, giữ slot cho tới khi người có quyền xác nhận worker của đúng fence đã dừng **và** tác động đã đối soát. Giải phóng slot không tự requeue. Chỉ retry lỗi tạm thời trước tác động, tối đa 1 lần thử lại (tổng <=2 và <= max_attempts), còn ngân sách, có backoff tối thiểu 60s, còn định danh hợp lệ và deadline. Không retry lỗi xác thực, chi phí/quota, sai SHA hoặc kết quả chưa rõ. Sau kill switch không tự retry lần thử đã dừng. Không chạy lại khi COMPLETED/CANCELLED, revision cũ hoặc lease cũ. Sửa/hủy khi active phải thu hồi, dừng và xác minh trước khi revision mới nhận việc. Nếu adapter không bảo đảm fence, kill switch, theo dõi hạn mức và đối soát thật thì không bật.
Task transitions: QUEUED -> RECEIVED -> WORKING -> REVIEW -> COMPLETED; BLOCKED từ các trạng thái chưa hoàn thành, CANCELLED theo quyền; BLOCKED -> QUEUED chỉ khi Director duyệt xử lý blocker, an toàn retry và revision còn hiệu lực. // Offline reference only. Single process; no network, persistent store, actual auth or side effects. Context booleans must come from a verified adapter in a future implementation; never trust payload flags. Tests are synthetic, not runtime acceptance.
const activeStates=["RECEIVED","WORKING","REVIEW"];
const retryBackoffMs=60000;
function newModel(){return {enabled:false,tasks:Object.create(null),slots:Object.create(null)}}

function addTask(s,t){
  if(!t||typeof t.task_id!=="string"||!t.task_id||!Number.isInteger(t.revision)||t.revision<1)return "INVALID";
  const old=s.tasks[t.task_id];
  if(old&&(!["CANCELLED","COMPLETED"].includes(old.state)||t.revision!==old.task.revision+1))return "REVISION_BLOCKED";
  s.tasks[t.task_id]={task:JSON.parse(JSON.stringify(t)),state:"QUEUED",attempts:0,fence:old?old.fence+1:0,leaseUntil:0,identityUntil:0,retryAt:0,worker:null,steps:new Set(),callsUsed:0,blocker:null};
  return "QUEUED";
}

// The synthetic adapter stamps its current observation at each boundary. A previous observation is stale.
function matchesReference(value,expected){return typeof value==="string"&&value.trim()!==""&&value===expected}

function contextBlocker(t,ctx,now){
  if(!ctx||!Number.isFinite(now)||ctx.verifiedAt!==now||ctx.eventVerified!==true||ctx.issuerAuthorized!==true||ctx.approvalVerified!==true||ctx.envelopeVerified!==true||ctx.eventKind!=="task"||!matchesReference(ctx.issuerId,t.issuer_id)||!matchesReference(ctx.approvalRef,t.approval_ref))return "UNAUTHORIZED";
  if(ctx.identityLevel!=="system_verified"||ctx.identityRevoked!==false||!Number.isFinite(ctx.identityExpiry)||ctx.identityExpiry<=now)return "IDENTITY_UNVERIFIED";
  if(!matchesReference(ctx.memberId,t.recipient_id)||!matchesReference(ctx.bindingRef,t.recipient_binding_ref)||!matchesReference(ctx.workspaceId,t.workspace_id)||!matchesReference(ctx.mode,t.mode)||!matchesReference(ctx.projectRef,t.project_ref))return "IDENTITY_MISMATCH";
  if(ctx.accessVerified!==true||!matchesReference(ctx.coreSha,t.core_sha)||!matchesReference(ctx.sourceSha,t.source_sha)||ctx.dependenciesReady!==true)return "ACCESS_OR_SOURCE_BLOCKED";
  if(ctx.quotaKnown!==true||ctx.quotaEnough!==true||ctx.zeroNewSpend!==true||t.new_spend_cap!==0)return "LIMIT_BLOCKED";
  return null;
}

function claim(s,id,rev,ctx,now){
  const r=s.tasks[id];if(!s.enabled)return "DISABLED";
  if(!r||r.task.revision!==rev)return "STALE_REVISION";
  const t=r.task;
  const blocker=contextBlocker(t,ctx,now);
  if(blocker)return blocker;
  if(!Number.isFinite(t.expires_at)||now>=t.expires_at||!Number.isSafeInteger(t.max_attempts)||t.max_attempts<1||!Number.isSafeInteger(t.max_runtime_seconds)||t.max_runtime_seconds<1||!Number.isSafeInteger(t.max_tool_calls)||t.max_tool_calls<1)return "INVALID_LIMIT";
  expireLease(s,id,rev,now);
  if(r.state!=="QUEUED"||s.slots[t.recipient_id])return "BUSY";
  if(r.callsUsed>=t.max_tool_calls)return "LIMIT_BLOCKED";
  if(now<r.retryAt||r.attempts>=Math.min(2,t.max_attempts))return "RETRY_BLOCKED";
  r.attempts++;r.fence++;r.worker=t.recipient_id;
  r.identityUntil=ctx.identityExpiry;
  r.leaseUntil=Math.min(r.identityUntil,t.expires_at,now+t.max_runtime_seconds*1000);
  r.state="RECEIVED";s.slots[r.worker]=id;
  return {state:r.state,fence:r.fence,leaseUntil:r.leaseUntil};
}

function block(r,reason){
  if(r.state!=="BLOCKED"){
    r.fence++;
    r.state="BLOCKED";
    r.blocker=reason;
  }
  return r.state;
}

function expireLease(s,id,rev,now){
  const r=s.tasks[id];
  if(!r||r.task.revision!==rev)return "STALE_REVISION";
  if(!Number.isFinite(now))return "INVALID";
  if(activeStates.includes(r.state)&&now>=r.leaseUntil){
    return block(r,now>=r.identityUntil?"IDENTITY_UNVERIFIED":"LEASE_EXPIRED");
  }
  return r.state;
}

function boundary(s,id,rev,fence,ctx,now){
  const r=s.tasks[id];
  if(!s.enabled||!r||r.task.revision!==rev||r.fence!==fence||!activeStates.includes(r.state))return "FENCED";
  if(expireLease(s,id,rev,now)==="BLOCKED")return "FENCED";
  const blocker=contextBlocker(r.task,ctx,now);
  if(blocker){block(r,blocker);return blocker}
  r.identityUntil=Math.min(r.identityUntil,ctx.identityExpiry);
  r.leaseUntil=Math.min(r.leaseUntil,r.identityUntil);
  return r;
}

function releaseSlot(s,r){
  if(r.worker!==null)delete s.slots[r.worker];
  r.worker=null;
}

function advance(s,id,rev,fence,to,ctx,now){
  const r=boundary(s,id,rev,fence,ctx,now);
  if(typeof r==="string")return r;
  const next={RECEIVED:"WORKING",WORKING:"REVIEW",REVIEW:"COMPLETED"};
  if(next[r.state]!==to)return "INVALID_TRANSITION";
  r.state=to;
  if(to==="COMPLETED"){releaseSlot(s,r);r.fence++}
  return to;
}

// One reservation per simulated model/tool invocation, even if it fails or never returns.
function reserveCall(s,id,rev,fence,kind,ctx,now){
  const r=boundary(s,id,rev,fence,ctx,now);
  if(typeof r==="string")return r;
  if(!["RECEIVED","WORKING"].includes(r.state))return "FENCED";
  if(!["model","tool"].includes(kind))return "INVALID";
  if(r.callsUsed>=r.task.max_tool_calls){block(r,"LIMIT_BLOCKED");return "LIMIT_BLOCKED"}
  r.callsUsed++;
  return "RESERVED";
}

function step(s,id,rev,fence,key,ctx,now){
  const r=boundary(s,id,rev,fence,ctx,now);
  if(typeof r==="string")return r;
  if(r.state!=="WORKING")return "FENCED";
  if(ctx.idempotentTarget!==true)return block(r,"EFFECT_UNSAFE");
  if(typeof key!=="string"||!key)return "INVALID";
  if(r.steps.has(key))return "DUPLICATE";
  r.steps.add(key);
  return "SIMULATED";
}

function revoke(s,id,rev,ctx,now){
  const r=s.tasks[id];
  if(!r||r.task.revision!==rev||!ctx||ctx.authorized!==true||!Number.isFinite(now)||ctx.verifiedAt!==now)return "UNAUTHORIZED";
  if(["COMPLETED","CANCELLED"].includes(r.state))return "TERMINAL";
  expireLease(s,id,rev,now);
  if(ctx.fence!==r.fence)return "FENCED";
  block(r,"STOP_REQUESTED");
  if(ctx.workerStopped!==true||ctx.effectsReconciled!==true)return r.state;
  releaseSlot(s,r);
  if(ctx.cancel===true){r.state="CANCELLED";return r.state}
  const t=r.task;
  const retryAt=now+retryBackoffMs;
  if(!s.enabled||ctx.retry!==true||ctx.failureKind!=="transient"||ctx.transientBeforeEffect!==true||r.steps.size!==0||
     !["STOP_REQUESTED","LEASE_EXPIRED"].includes(r.blocker)||
     r.callsUsed>=t.max_tool_calls||r.attempts>=Math.min(2,t.max_attempts)||
     retryAt>=t.expires_at||contextBlocker(t,ctx.executionContext,now)||
     retryAt>=ctx.executionContext.identityExpiry)return r.state;
  r.retryAt=retryAt;
  r.blocker=null;
  r.state="QUEUED";
  return r.state;
}

function stop(s){
  s.enabled=false;
  for(const r of Object.values(s.tasks)){
    if(activeStates.includes(r.state))block(r,"KILL_SWITCH");
  }
}
